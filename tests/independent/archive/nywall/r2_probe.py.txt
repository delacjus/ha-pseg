# Execute the exact captcha form methods in an isolated AST harness: no HA runtime.
source = (ROOT / 'custom_components/pseg/config_flow.py').read_text()
tree = ast.parse(source)
names = {'_helper_url', '_async_show_captcha_form', 'async_step_captcha'}
methods = [n for n in ast.walk(tree) if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef)) and n.name in names]
module = ast.fix_missing_locations(ast.Module(body=methods, type_ignores=[]))
env = {'_CAPTCHA_STEP_ID': 'captcha', 'HELPER_URL_PATH': '/api/pseg/helper.zip',
       'PORTAL_BASE_URL': 'https://portal.example', 'CONF_CAPTCHA_RESPONSE': 'captcha_response',
       'CONF_CAPTCHA_CANCEL': 'captcha_cancel', 'vol': SimpleNamespace(Schema=lambda x: x, Optional=lambda x, **kw: x),
       'TextSelector': lambda x: x, 'TextSelectorConfig': lambda **kw: kw,
       'TextSelectorType': SimpleNamespace(PASSWORD='password')}
exec(compile('from __future__ import annotations\n', '<annotations>', 'exec'), env)
exec(compile(module, '<captcha-methods>', 'exec', flags=__import__('__future__').annotations.compiler_flag), env)
class Harness:
    _helper_url = env['_helper_url']
    _async_show_captcha_form = env['_async_show_captcha_form']
    async_step_captcha = env['async_step_captcha']
    hass = SimpleNamespace(config=SimpleNamespace(internal_url=None, external_url=None))
    def _attempt_expired(self, pending):
        return False
    def async_show_form(self, **kwargs):
        return kwargs

for origin in ('user', 'reauth_confirm', 'reconfigure'):
    h = Harness()
    h._captcha_pending = {'step_id': origin}
    form = asyncio.run(h.async_step_captcha())
    assert form['step_id'] == 'captcha'
    for fname in ['strings.json', 'translations/en.json']:
        data = json.loads((ROOT / 'custom_components/pseg' / fname).read_text())

        rendered = desc.format(**form['description_placeholders'])
        assert not __import__('re').search(r'\{[^{}]+\}', rendered)
    print('CAPTCHA', origin, 'all four placeholders rendered')
