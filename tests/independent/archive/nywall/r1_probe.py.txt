# Avoid client initialization, requests sessions, and all authentication paths.
client = object.__new__(PsegClient)
DAY = date(2026, 10, 15)

def point(day, hour, value):
    wall = datetime(day.year, day.month, day.day, hour, tzinfo=UTC)
    label = day.strftime('%m/%d/%Y')
    return {'x': int(wall.timestamp() * 1000), 'y': value,
            'hs': {'start': label, 'end': label}}

BASE = [point(DAY, h, round(0.4 + (h % 7) * 0.13, 4)) for h in range(24)]

def run(name, points, totals, expected_complete=None, expected_reason=None):
    parsed = client._parse_hourly_chart(
        {'Data': {'series': [{'name': 'Meter #1', 'data': points}]}}
    )
    result = validate_hourly_days(
        parsed['hourly_records'], totals, datetime(2028, 1, 1, tzinfo=UTC),
        granularity=parsed['hourly_granularity'], days=totals,
    )
    checks = list(result.values())
    complete = sum(c.complete for c in checks)
    reasons = dict(Counter(c.reason for c in checks if c.reason))
    summary = dict(case=name, complete=complete, reasons=reasons,
                   raw_points=len(points), records=len(parsed['hourly_records']),
                   invalid=parsed['hourly_invalid_points'],
                   interval_failures=dict(Counter(r['interval_reason'] for r in
                       parsed['hourly_records'] if r['interval_reason'])))
    if parsed['hourly_records']:
        summary['utc_bounds'] = [parsed['hourly_records'][0]['start'].isoformat(),
                                 parsed['hourly_records'][-1]['start'].isoformat()]
    if expected_complete is not None:
        assert complete == expected_complete, summary
    if expected_reason:
        assert expected_reason in reasons, summary
    print(json.dumps(summary, sort_keys=True))
    return parsed, result

for name, transform in [

# Exact conversion vs. expected UTC sets over all 2026 days, including boundaries.
day = date(2026, 1, 1)
statuses = Counter()
for i in range(365):
    day = date(2026, 1, 1) + timedelta(days=i)
    expected = set(expected_hour_starts(day))
    got = set()
    for h in range(24):
        st, status = ny_wall_ms_to_utc(point(day, h, 1)['x'])
        statuses[status] += 1
        if st is not None:
            got.add(st)
            assert ny_day_bounds(day)[0] <= st < ny_day_bounds(day)[1]
    if day == date(2026, 11, 1):
        assert expected - got == {datetime(2026, 11, 1, 6, tzinfo=UTC)}
    else:
        assert got == expected, day
print('ALL_2026_CONVERSIONS', dict(statuses), '365 days checked')

# AST-check every fixed emitted validation/interval reason, excluding helper status.
emitted = set()
for f in ('statistics_hourly.py', 'pseg_client.py'):
    tree = ast.parse((ROOT / 'custom_components/pseg' / f).read_text())
    for n in ast.walk(tree):
        if isinstance(n, ast.keyword) and n.arg == 'reason' and isinstance(n.value, ast.Constant) and isinstance(n.value.value, str):
            emitted.add(n.value.value)
        if isinstance(n, ast.Assign) and any(isinstance(t, ast.Name) and t.id in ('point_reason', 'interval_failure', 'reason') for t in n.targets):
            for k in ast.walk(n.value):
                if isinstance(k, ast.Constant) and isinstance(k.value, str) and k.value in STATISTICS_INCOMPLETE_REASONS:
                    emitted.add(k.value)
assert emitted <= STATISTICS_INCOMPLETE_REASONS, emitted - STATISTICS_INCOMPLETE_REASONS
assert 'dst_ambiguous' in emitted
print('REASON_CLOSURE', sorted(emitted), 'PASS')

        n.value = ast.parse('ny_wall_ms_to_utc(timestamp)[0]', mode='eval').body
old_module = ast.fix_missing_locations(ast.Module(body=[old_method], type_ignores=[]))
env = dict(client_module.__dict__)
exec(compile(old_module, '<HEAD-gate-with-new-conversion>', 'exec'), env)
for kind in ('bare_same_end', 'aware_missing_end', 'date_start_wrong_aware_end', 'basic_aware_wrong_start'):

    for p in pts:

            p['hs'] = {'start': label, 'end': label}
        elif kind == 'aware_missing_end':
            p['hs'] = {'start': st.isoformat()}
        elif kind == 'date_start_wrong_aware_end':
            p['hs']['end'] = '2026-10-15T12:00:00-04:00'
        else:
            p['hs']['start'] = '20261015T230000-0400'
    payload = {'Data': {'series': [{'name': 'Meter #1', 'data': pts}]}}
    old = env['_parse_hourly_chart'](client, payload)
    new = client._parse_hourly_chart(payload)
    assert not any(r['interval_ok'] for r in old['hourly_records']), kind
    assert all(r['interval_ok'] for r in new['hourly_records']), kind
    print('OLD_GATE_CONTROL', kind, 'before=24 rejected; after=24 accepted')
